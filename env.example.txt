# ---------------------------------------------------------------------------
# ANTI-BOT SCORING (this branch) — read SECURITY_REVIEW.md §2 before tuning
# ---------------------------------------------------------------------------
# The game is deterministic and physics.js is public, so a script CAN solve a
# run. These settings are what stop a solved run from being banked.

# Absolute score ceiling. A run above it is rejected outright. Measured: adding
# +-1 step (~17ms) of timing noise to a solved 89-point run drops it to 17, and
# +-3 steps (~50ms) drops it to 3.5 — so triple-digit scores are machine-made.
# Lower it if your real players never get near it. Default 150.
# SCORE_HARD_CAP=150

# Progressive ceiling: a run may not beat the account's own best by more than
# SCORE_JUMP_MAX, and the absolute ceiling grows with verified-run count.
# A farmed alt-account therefore cannot open with a big score. Set
# SCORE_GATE_ENABLED=false to turn the whole progression gate off.
# SCORE_GATE_ENABLED=true
# SCORE_NEW_ACCOUNT_CAP=30     # first 15 verified runs
# SCORE_YOUNG_ACCOUNT_CAP=60   # runs 15..39
# SCORE_TRUSTED_CAP=150        # 40+ verified runs
# SCORE_JUMP_MAX=12            # max improvement over the account's own best
# SCORE_NEW_ACCOUNT_RUNS=15
# SCORE_YOUNG_ACCOUNT_RUNS=40

# Autopilot telemetry (removable-tap ratio + timing rigidity) is computed for
# every high score and listed at GET /internal/suspects, but never bans on its
# own — measurements showed it cannot separate a bot from an excellent player.
# SOLVER_MIN_SCORE=40    # score at which telemetry starts
# RIGIDITY_TRIALS=4
